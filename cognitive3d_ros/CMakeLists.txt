# Plain signature throughout: ament_target_dependencies uses the plain form,
# and CMake refuses to mix keyword and plain on one target.
target_link_libraries(cognitive3d_ros_component c3d_core ${OpenCV_LIBS}
  nlohmann_json::nlohmann_json ${TinyXML2_LIBRARIES})
target_include_directories(cognitive3d_ros_component PRIVATE ${TinyXML2_INCLUDE_DIRS})
ament_target_dependencies(cognitive3d_ros_component
  rclcpp rclcpp_lifecycle rclcpp_components ament_index_cpp lifecycle_msgs diagnostic_msgs
  rosgraph_msgs tf2 tf2_ros std_msgs geometry_msgs sensor_msgs nav_msgs tf2_msgs action_msgs
  cv_bridge cognitive3d_ros_msgs)
target_compile_options(cognitive3d_ros_component PRIVATE -Wall -Wextra)

rclcpp_components_register_nodes(cognitive3d_ros_component "cognitive3d_ros::Cognitive3DNode")

# Hand-written rather than generated by rclcpp_components, so that a node its parameters cannot
# construct says which parameter and how to fix it.
add_executable(cognitive3d_node src/node_main.cpp)
target_link_libraries(cognitive3d_node cognitive3d_ros_component)
ament_target_dependencies(cognitive3d_node rclcpp)
target_compile_options(cognitive3d_node PRIVATE -Wall -Wextra)
install(TARGETS cognitive3d_node DESTINATION lib/${PROJECT_NAME})

# The preflight command. Constructs the REAL node in dry_run and reports what it resolved, so it
# cannot drift out of step with the node the way a re-derivation would. It becomes
# cognitive3d_ros_nav2's doctor when that package is installed, as the launch file runs its node.
add_executable(doctor tools/doctor_main.cpp)
target_link_libraries(doctor cognitive3d_ros_component)
ament_target_dependencies(doctor rclcpp)
install(TARGETS doctor DESTINATION lib/${PROJECT_NAME})

install(TARGETS cognitive3d_ros_component
  ARCHIVE DESTINATION lib
  LIBRARY DESTINATION lib
  RUNTIME DESTINATION bin)
# Headers under include/<package>, the layout every package uses from Humble on, so this package
# never shares an include directory with another in a merged prefix. core's headers go with them:
# node.hpp includes them, and core installs nothing of its own as a subproject.
install(DIRECTORY include/ DESTINATION include/${PROJECT_NAME})
install(DIRECTORY ${CMAKE_CURRENT_SOURCE_DIR}/../core/include/ DESTINATION include/${PROJECT_NAME})
install(DIRECTORY config launch DESTINATION share/${PROJECT_NAME})

# The integrator tools, run as `ros2 run cognitive3d_ros <tool>`. They import c3d_api and each
# other from their own directory, so the package goes beside them. The test harnesses that share
# tools/ with them are not installed.
set(C3D_TOOLS_DIR ${CMAKE_CURRENT_SOURCE_DIR}/../tools)
install(PROGRAMS
  ${C3D_TOOLS_DIR}/fetch_application_key.py
  ${C3D_TOOLS_DIR}/verify_gaze_contract.py
  ${C3D_TOOLS_DIR}/verify_dynamics_level.py
  ${C3D_TOOLS_DIR}/export_world_gltf.sh
  ${C3D_TOOLS_DIR}/gltf_bbox.py
  ${C3D_TOOLS_DIR}/collada_zup_repair.py
  DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY ${C3D_TOOLS_DIR}/c3d_api DESTINATION lib/${PROJECT_NAME}
  FILES_MATCHING PATTERN "*.py" PATTERN "__pycache__" EXCLUDE)
install(FILES LICENSE ${CMAKE_CURRENT_SOURCE_DIR}/../THIRD_PARTY_NOTICES.md
  DESTINATION share/${PROJECT_NAME})
